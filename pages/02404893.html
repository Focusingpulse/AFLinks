ȻɊɂɇɄȿȼɂɑ (Ȼɟɥɨɪɭɫɫɤɢɣ ɝɨɫɭɞɚɪɫɬɜɟɧɧɵɣ ɭɧɢɜɟɪɫɢɬɟɬ, Ɇɢɧɫɤ); ɇ.ȼ. ȼȺȻɂɓȿȼɂɑ (ɉɨɥɨɰɤɢɣ ɝɨɫɭɞɚɪɫɬɜɟɧɧɵɣ ɭɧɢɜɟɪɫɢɬɟɬ) ɂɫɫɥɟɞɨɜɚɧɨ ɪɚɫɩɪɟɞɟɥɟɧɢɟ ɪɚɞɢɨɧɭɤɥɢɞɨɜ ɩɨ ɪɚɡɥɢɱɧɵɦ ɤɚɪɬɪɢɞɠɚɦ ɜ ɩɪɨɰɟɫɫɟ ɩɪɨɢɡɜɨɞɫɬɜɚ F-ɮɬɨɪɞɟɡɨɤɫɢɝɥɸɤɨɡɵ ɢ 18F-ɯɨɥɢɧɚ ɫ ɢɫɩɨɥɶɡɨɜɚɧɢɟɦ ɰɢɤɥɨɬɪɨɧɚ IBA Cyclone 18/9 ɫ ɰɟɥɶɸ ɪɚɡɪɚɛɨɬɤɢ ɧɚɭɱɧɨ ɨɛɨɫɧɨɜɚɧɧɨɣ ɤɨɧɰɟɩɰɢɢ ɞɨɥɝɨɜɪɟɦɟɧɧɨɝɨ ɯɪɚɧɟɧɢɹ ɪɚɞɢɨɚɤɬɢɜɧɵɯ ɨɬɯɨɞɨɜ ɪɚɞɢɨɮɚɪɦɩɪɟɩɚɪɚɬɨɜ ɞɥɹ ɩɨɡɢɬɪɨɧɧɨ-ɷɦɢɫɫɢɨɧɧɨɣ ɬɨɦɨɝɪɚɮɢɢ. Ɉɩɪɟɞɟɥɟɧ ɢɡɨɬɨɩɧɵɣ ɫɨɫɬɚɜ ɞɨɥɝɨɠɢɜɭɳɢɯ (t1/2 = 10–312 ɫɭɬ) ɪɚɞɢɨɧɭɤɥɢɞɨɜ. ɉɨɤɚɡɚɧɨ, ɱɬɨ ɫɪɨɤ ɯɪɚɧɟɧɢɹ ɤɚɪɬɪɢɞɠɟɣ QMA ɞɨɥɠɟɧ ɫɨɫɬɚɜɥɹɬɶ ɧɟ ɦɟɧɟɟ ɩɨɥɭɝɨɞɚ. Ɋɚɫɫɦɨɬɪɟɧɵ ɪɚɡɥɢɱɧɵɟ ɦɟɯɚɧɢɡɦɵ ɩɨɩɚɞɚɧɢɹ ɪɚɞɢɨɧɭɤɥɢɞɨɜ ɦɟɬɚɥɥɨɜ ɩɟɪɟɦɟɧɧɨɣ ɜɚɥɟɧɬɧɨɫɬɢ ɜ ɤɚɪɬɪɢɞɠɢ ɩɪɢ ɫɢɧɬɟɡɟ ɪɚɞɢɨɮɚɪɦɩɪɟɩɚɪɚɬɨɜ ɧɚ ɨɫɧɨɜɟ 18F. 18 Ʉɥɸɱɟɜɵɟ ɫɥɨɜɚ: ɪɚɞɢɨɮɚɪɦɩɪɟɩɚɪɚɬɵ, ɪɚɞɢɨɧɭɤɥɢɞɵ, 18F-ɮɬɨɪɞɟɡɨɤɫɢɝɥɸɤɨɡɚ, 18F-ɯɨɥɢɧ. ɉɨɡɢɬɪɨɧɧɨ-ɷɦɢɫɫɢɨɧɧɚɹ ɬɨɦɨɝɪɚɮɢɹ (ɉɗɌ) – ɫɨɜɪɟɦɟɧɧɵɣ ɦɟɬɨɞ ɦɟɞɢɰɢɧɫɤɨɣ ɜɢɡɭɚɥɢɡɚɰɢɢ, ɩɨɡɜɨɥɹɸɳɢɣ ɩɨɥɭɱɚɬɶ in vivo ɢɧɮɨɪɦɚɰɢɸ ɨ ɧɚɪɭɲɟɧɢɹɯ ɮɢɡɢɨɥɨɝɢɱɟɫɤɢɯ ɢ ɛɢɨɯɢɦɢɱɟɫɤɢɯ ɩɪɨɰɟɫɫɨɜ ɧɚ ɦɨɥɟɤɭɥɹɪɧɨɦ ɭɪɨɜɧɟ. ɇɚ ɛɚɡɟ Ɋɟɫɩɭɛɥɢɤɚɧɫɤɨɝɨ ɧɚɭɱɧɨ-ɩɪɚɤɬɢɱɟɫɤɨɝɨ ɰɟɧɬɪɚ ɨɧɤɨɥɨɝɢɢ ɢ ɦɟɞɢɰɢɧɫɤɨɣ ɪɚɞɢɨɥɨɝɢɢ ɢɦ. ɇ.ɇ. Ⱥɥɟɤɫɚɧɞɪɨɜɚ ɫ ɧɨɹɛɪɹ 2015 ɝ. ɜɩɟɪɜɵɟ ɜ Ɋɟɫɩɭɛɥɢɤɟ Ȼɟɥɚɪɭɫɶ ɡɚɪɚɛɨɬɚɥ Ɋɟɫɩɭɛɥɢɤɚɧɫɤɢɣ ɰɟɧɬɪ ɩɨɡɢɬɪɨɧɧɨ-ɷɦɢɫɫɢɨɧɧɨɣ ɬɨɦɨɝɪɚɮɢɢ (ɉɗɌ-ɰɟɧɬɪ). ȿɝɨ ɫɨɡɞɚɧɢɟ ɩɨɡɜɨɥɢɥɨ ɭɥɭɱɲɢɬɶ ɪɟɡɭɥɶɬɚɬɵ ɥɟɱɟɧɢɹ ɩɚɰɢɟɧɬɨɜ ɫ ɨɧɤɨɩɚɬɨɥɨɝɢɟɣ; ɫɨɤɪɚɬɢɬɶ ɷɤɨɧɨɦɢɱɟɫɤɢɟ ɪɚɫɯɨɞɵ ɧɚ ɩɪɨɜɟɞɟɧɢɟ ɥɟɤɚɪɫɬɜɟɧɧɨɣ ɬɟɪɚɩɢɢ; ɭɜɟɥɢɱɢɬɶ ɞɨɥɸ ɮɢɧɚɧɫɨɜɨ-ɫɛɟɪɟɝɚɸɳɢɯ ɨɩɟɪɚɰɢɣ; ɢɫɤɥɸɱɢɬɶ ɧɟɨɛɯɨɞɢɦɨɫɬɶ ɧɚɩɪɚɜɥɟɧɢɹ ɩɚɰɢɟɧɬɨɜ ɞɥɹ ɞɢɚɝɧɨɫɬɢɤɢ ɢ ɥɟɱɟɧɢɹ ɡɚ ɪɭɛɟɠ. ȼ ɉɗɌ-ɢɫɫɥɟɞɨɜɚɧɢɹɯ ɜ ɤɚɱɟɫɬɜɟ ɪɚɞɢɨɚɤɬɢɜɧɨɣ ɦɟɬɤɢ ɢɫɩɨɥɶɡɭɸɬɫɹ ɤɨɪɨɬɤɨɠɢɜɭɳɢɟ ɪɚɞɢɨɢɡɨɬɨɩɵ ɫ ɩɨɡɢɬɪɨɧɧɵɦ ɪɚɫɩɚɞɨɦ.
